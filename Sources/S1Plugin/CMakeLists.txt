# S1Plugin — Arcade Ruins as a JUCE plugin (X2-1, ADR-054/055/057/058/072).
#
# Built only with -DS1_BUILD_PLUGIN=ON, so the engine's CI stays a one-minute build:
#
#   cmake -S . -B build/plugin -DS1_BUILD_PLUGIN=ON && cmake --build build/plugin --target ArcadeRuins_All
#
# JUCE is FETCHED at configure time, pinned by tag AND commit. It is never copied into this
# repository: section 1.17 of the JUCE 9 EULA does not let the Framework be distributed on its
# own, and Scripts/publish-public.sh mirrors this tree with `git archive` (ADR-057).

include(FetchContent)

# 9.0.2. The commit is what is fetched; a tag can be moved, a commit cannot.
set(S1_JUCE_TAG 9.0.2)
set(S1_JUCE_COMMIT 72782788ce18c2d4d760b28e0921d6ffc6431102)

FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG ${S1_JUCE_COMMIT}
    GIT_SHALLOW FALSE          # a bare commit cannot be fetched shallow from every git version
    GIT_PROGRESS TRUE)
FetchContent_MakeAvailable(JUCE)

# Parameter IDs are GENERATED from S1Parameter.h, never typed (ADR-058, ADR-073): the ID of a
# parameter is the name of its enum case. Each `name = number,` line of the enum becomes
# `S1_PARAMETER_NAME(name, number)` in S1ParameterNames.inc; S1ParameterCatalog.cpp expands it and
# static_asserts every number against the enum itself. Tests/Plugin/Fixtures/parameter-ids.txt is
# the frozen list a renamed case fails against.
set(S1_PARAMETER_HEADER ${CMAKE_SOURCE_DIR}/Sources/S1Engine/S1Parameter.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${S1_PARAMETER_HEADER})
file(STRINGS ${S1_PARAMETER_HEADER} S1_PARAMETER_LINES REGEX "^ +[A-Za-z][A-Za-z0-9_]* = [0-9]+,?")
set(S1_PARAMETER_NAMES "// GENERATED by Sources/S1Plugin/CMakeLists.txt from Sources/S1Engine/S1Parameter.h. Do not edit.\n")
set(S1_PARAMETER_NAME_COUNT 0)
foreach(line IN LISTS S1_PARAMETER_LINES)
    string(REGEX MATCH "([A-Za-z][A-Za-z0-9_]*) = ([0-9]+)" _ "${line}")
    if(NOT CMAKE_MATCH_1 STREQUAL "S1ParameterCount")
        string(APPEND S1_PARAMETER_NAMES "S1_PARAMETER_NAME(${CMAKE_MATCH_1}, ${CMAKE_MATCH_2})\n")
        math(EXPR S1_PARAMETER_NAME_COUNT "${S1_PARAMETER_NAME_COUNT} + 1")
    endif()
endforeach()
if(NOT S1_PARAMETER_NAME_COUNT EQUAL 150)
    message(FATAL_ERROR "S1Parameter.h: expected 150 parameters, read ${S1_PARAMETER_NAME_COUNT} — has the enum's layout changed?")
endif()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/S1ParameterNames.inc CONTENT "${S1_PARAMETER_NAMES}" @ONLY)
target_include_directories(ArcadeRuins PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)

# The oscillator tables, linked in: the same 54 JSON files the Mac bundle carries, read through
# s1::Wavetables::load(Source) (ADR-070). Listed by glob on purpose — the index file inside says
# which 52 tables must exist, and the loader refuses by name when one is missing.
file(GLOB_RECURSE S1_WAVETABLE_JSON CONFIGURE_DEPENDS
    ${CMAKE_SOURCE_DIR}/Sources/SynthOneCore/DSP/BandlimitedWavetables/*.json)
juce_add_binary_data(ArcadeRuinsData
    HEADER_NAME S1BinaryData.h
    NAMESPACE S1BinaryData
    SOURCES ${S1_WAVETABLE_JSON})

# X2-9 (ADR-080): the thirteen factory banks, the Mac app's own files, linked in the same way.
# Read-only because they are the binary; Tests/Plugin/PluginPresetTests holds them to the
# repository's files byte for byte.
file(GLOB S1_FACTORY_BANK_JSON CONFIGURE_DEPENDS
    ${CMAKE_SOURCE_DIR}/Sources/SynthOneCore/Presets/Data/*.json)
list(LENGTH S1_FACTORY_BANK_JSON S1_FACTORY_BANK_COUNT)
if(NOT S1_FACTORY_BANK_COUNT EQUAL 13)
    message(FATAL_ERROR "Expected 13 factory bank files, found ${S1_FACTORY_BANK_COUNT}")
endif()
juce_add_binary_data(ArcadeRuinsPresets
    HEADER_NAME S1PresetData.h
    NAMESPACE S1PresetData
    SOURCES ${S1_FACTORY_BANK_JSON})
